// Corgi's Pi integration. Managed by `corgi integration install pi`.
import { spawn } from "node:child_process";
import { randomUUID, createHash } from "node:crypto";
import { createServer } from "node:net";
import { chmodSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
export default function (pi) {
  const owner = process.env.CORGI_PI_OWNER;
  if (owner && owner !== String(process.pid)) return;
  let turn = "", result = "completed", escaped = false, prompts = 0, prompt = "", removeInput, compacting = false;
  const inputs = new Map();
  let server;
  const active = ctx => ctx.mode === "tui" && process.env.CORGI_PANE && process.env.CORGI_SOCKET && process.env.CORGI_BIN;
  const report = (ctx, event, extra = {}) => {
    if (!active(ctx)) return Promise.resolve();
    const data = { event, session: ctx.sessionManager.getSessionId(), transcript: ctx.sessionManager.getSessionFile() || "", cwd: ctx.cwd,
      turn, account: process.env.PI_CODING_AGENT_DIR ?? join(process.env.HOME, ".pi/agent"), model: ctx.model ? ctx.model.provider + "/" + ctx.model.id : "", ...extra };
    return new Promise(resolve => {
      const child = spawn(process.env.CORGI_BIN, ["hook", "pi"], { stdio: ["pipe", "ignore", "ignore"], timeout: 2200 });
      child.on("error", resolve); child.on("close", resolve); child.stdin.on("error", () => {});
      child.stdin.end(JSON.stringify(data));
    });
  };
  pi.on("session_start", async (_e, ctx) => {
    if (!active(ctx)) return;
    process.env.CORGI_PI_OWNER = String(process.pid);
    removeInput = ctx.ui.onTerminalInput(data => { if (turn && !prompts && data === "\x1b") escaped = true; });
    const root = process.env.PI_CODING_AGENT_DIR ?? join(process.env.HOME, ".pi/agent");
    const session = ctx.sessionManager.getSessionId();
    const hash = createHash("sha256").update(root+"\n"+session).digest("hex").slice(0,24);
    const socketPath = join(tmpdir(), `corgi-pi-${process.getuid?.() ?? 0}-${hash}-${process.pid}.sock`);
    // Corgi renames the session and sends it prompts through this socket.
    server = createServer(socket => {
      let bytes = "";
      socket.setEncoding("utf8"); socket.setTimeout(1500, () => socket.destroy()); socket.on("error", () => {});
      socket.on("data", chunk => {
        bytes += chunk;
        if(bytes.length > 8388608) return socket.destroy();
        if(!bytes.includes("\n")) return;
        try {
          const r = JSON.parse(bytes);
          if(r.session !== ctx.sessionManager.getSessionId()) return socket.end('{"ok":false}\n');
          if(r.prompt !== undefined) {
            // Pi refuses a prompt while compacting; Corgi types it instead, and the editor queues it.
            if(typeof r.prompt !== "string" || !r.prompt || compacting) return socket.end('{"ok":false}\n');
            pi.sendUserMessage(r.prompt, { deliverAs: "steer" });
            return socket.end('{"ok":true,"prompted":true}\n');
          }
          if(r.name !== undefined) {
            if(typeof r.name !== "string" || r.name.length > 160 || /[\x00-\x1f]/.test(r.name)) return socket.end('{"ok":false}\n');
            if(r.only_unnamed && pi.getSessionName()) return socket.end('{"ok":false,"exists":true}\n');
            pi.setSessionName(r.name);
          }
          socket.end('{"ok":true,"automatic":true,"prompts":true}\n');
        } catch { socket.destroy(); }
      });
    });
    server.on("error", () => {});
    server.listen(socketPath, () => { try { chmodSync(socketPath,0o600); } catch {} });
    await report(ctx, "SessionStart");
  });
  pi.on("before_agent_start", (e, ctx) => { prompt = active(ctx) && e.messageType !== "custom" ? e.prompt || "" : ""; });
  pi.on("agent_start", async (_e, ctx) => {
    compacting = false;
    if (!active(ctx) || turn) return;
    turn = randomUUID(); result = "completed"; escaped = false;
    const opening = pi.getSessionName() ? undefined : ctx.sessionManager.getBranch().find(e => e.type === "message" && e.message.role === "user");
    const content = opening?.message.content ?? (pi.getSessionName() ? "" : prompt);
    const text = typeof content === "string" ? content : (content || []).filter(b => b.type === "text").map(b => b.text).join("\n");
    const bytes = Buffer.from(text);
    let end = Math.min(bytes.length, 2048);
    while(end > 0 && (bytes[end] & 0xc0) === 0x80) end--;
    await report(ctx, "UserPromptSubmit", { initial_request: bytes.subarray(0,end).toString() });
  });
  pi.on("message_end", (e, ctx) => {
    if (!active(ctx) || e.message.role !== "assistant") return;
    result = e.message.stopReason === "aborted" ? "interrupted" : ["error", "length"].includes(e.message.stopReason) ? "failed" : "completed";
    const usage = ctx.getContextUsage();
    pi.appendEntry("corgi-context", { model: ctx.model?.id, window: ctx.model?.contextWindow, used: usage?.tokens });
  });
  pi.on("tool_execution_start", (e, ctx) => { inputs.set(e.toolCallId, e.args); return report(ctx, "PreToolUse", { tool: e.toolName, input: e.args }); });
  pi.on("tool_execution_end", (e, ctx) => { const input = inputs.get(e.toolCallId); inputs.delete(e.toolCallId); return report(ctx, e.isError ? "PostToolUseFailure" : "PostToolUse", { tool: e.toolName, input }); });
  pi.on("ui_prompt_start", (e, ctx) => { prompts++; return report(ctx, "PermissionRequest", { tool: e.title || e.kind }); });
  pi.on("ui_prompt_end", (_e, ctx) => { prompts = Math.max(0, prompts-1); return prompts ? undefined : report(ctx, turn ? "PostToolUse" : "Idle"); });
  pi.on("session_before_compact", (_e, ctx) => { compacting = true; return report(ctx, "PreCompact"); });
  pi.on("session_compact", (_e, ctx) => { compacting = false; return report(ctx, turn ? "PostCompact" : "Idle"); });
  pi.on("session_compact_failed", (_e, ctx) => { compacting = false; return report(ctx, turn ? "PostCompact" : "Idle"); });
  pi.on("agent_settled", async (e, ctx) => {
    if (!active(ctx) || !turn) return;
    const outcome = e.aborted === true || (e.aborted === undefined && escaped) ? "interrupted" : result;
    // The receipt follows the hook so its timestamp passes Corgi's freshness check.
    await report(ctx, "Stop", { outcome });
    pi.appendEntry("corgi-settled", { session: ctx.sessionManager.getSessionId(), turn, outcome });
    turn = "";
  });
  pi.on("session_shutdown", async (_e, ctx) => { removeInput?.(); if(server) await new Promise(resolve => server.close(resolve)); await report(ctx, "SessionEnd"); });
}
